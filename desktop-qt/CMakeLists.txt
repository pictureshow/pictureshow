cmake_minimum_required(VERSION 3.22)
project(PictureshowQtProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui GuiPrivate Quick Qml Test)
qt_add_executable(qt-proof src/main.cpp src/player.cpp src/surface.cpp)
target_include_directories(qt-proof PRIVATE include src)
target_link_libraries(qt-proof PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::Quick Qt6::Qml Qt6::Test)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(qt-proof PRIVATE src/mac.mm)
  target_link_libraries(qt-proof PRIVATE "-framework AppKit" "-framework Metal" "-framework QuartzCore" "-framework IOSurface" "-framework OpenGL")
elseif(WIN32)
  target_link_libraries(qt-proof PRIVATE d3d11 dxgi)
endif()
qt_add_resources(qt-proof qml PREFIX "/" FILES Main.qml)
if(WIN32)
  qt_add_executable(hdr-api-state src/hdr_api.cpp)
  target_link_libraries(hdr-api-state PRIVATE Qt6::Core d3d11 dxgi)
endif()
add_subdirectory(core)
